async function t22LoadSplitFull(manifestName){
  const mr=await fetch(`${ROOT}/truth/full/${manifestName}`,{cache:'no-store'});if(!mr.ok)throw new Error(`${manifestName}: ${mr.status}`);
  const manifest=await mr.json(),parts=manifest.parts||[];if(!parts.length)throw new Error(`${manifestName}: aucune partie`);
  const texts=await Promise.all(parts.map(async name=>{const r=await fetch(`${ROOT}/truth/full/${name}`,{cache:'no-store'});if(!r.ok)throw new Error(`${name}: ${r.status}`);return r.text()}));
  const raw=await truthFullInflateBase64(texts.join('').replace(/\s+/g,''));
  const rows=Array.isArray(raw)?raw:(raw?.entries||[]);if(!Array.isArray(rows))throw new Error(`${manifestName}: contenu invalide`);
  const paid=rows.filter(x=>x&&x.id&&x.name&&Number.isFinite(Number(x.cost))&&Number(x.cost)>0&&Number(x.cost)<=3);
  if(Number.isInteger(manifest.entries)&&paid.length!==manifest.entries)throw new Error(`${manifestName}: ${paid.length}/${manifest.entries} entrées`);
  if(new Set(paid.map(x=>x.id)).size!==paid.length)throw new Error(`${manifestName}: IDs dupliqués`);
  return paid;
}

function t22WhenMatches(t){
  const when=t?.when;if(!when||typeof when!=='object'||Array.isArray(when))return null;
  const pairs=Object.entries(when);if(!pairs.length)return null;
  return pairs.every(([key,want])=>{
    const got=state.truth.choices?.[key];
    return Array.isArray(want)?want.includes(got):want===got;
  });
}
