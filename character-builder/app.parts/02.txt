function truthTalentVisible(t){
  const n=state.truth.nature,g=normText(t.group);
  if(n==='mage')return true;
  if(n==='humain'){
    const tr=state.truth.choices?.hunterTradition||'aucune';
    if(tr==='aucune')return false;
    return groupHas(g,truthNeedles.humain[tr]);
  }
  if(n==='vampire'){
    if(g.includes('talents communs de vampire'))return true;
    return groupHas(g,truthNeedles.vampire[state.truth.choices?.court])||groupHas(g,truthNeedles.vampire[state.truth.choices?.blood]);
  }
  if(n==='garou'){
    if(g.includes('talents communs de garou')||g.includes('maitrise d un sang')||g.includes('eveiller d autres sangs'))return true;
    return groupHas(g,truthNeedles.garou[state.truth.choices?.pelage])||groupHas(g,truthNeedles.garou[state.truth.choices?.blood]);
  }
  if(n==='khinae')return groupHas(g,truthNeedles.khinae[state.truth.choices?.lineage]);
  if(n==='daemon'){
    if(g.includes('talents communs de nature')||g.includes('origine de l elu'))return true;
    return groupHas(g,truthNeedles.daemon[state.truth.choices?.function])||groupHas(g,truthNeedles.daemon[state.truth.choices?.divinity]);
  }
  if(n==='angelus'){
    if(g.includes('talents communs')||g.includes('progression de transcendance'))return true;
    return groupHas(g,truthNeedles.angelus[state.truth.choices?.angelNature])||groupHas(g,truthNeedles.angelus[state.truth.choices?.sephirah]);
  }
  if(n==='aseryn'){
    if(g.includes('routes communes aserynes'))return true;
    return groupHas(g,truthNeedles.aseryn[state.truth.choices?.origin])||groupHas(g,truthNeedles.aseryn[state.truth.choices?.tradition])||groupHas(g,truthNeedles.aseryn[state.truth.choices?.seratheenTradition]);
  }
  if(n==='exile'){
    return groupHas(g,truthNeedles.exile[state.truth.choices?.people])||groupHas(g,truthNeedles.exile[state.truth.choices?.network]);
  }
  if(n==='extral'){
    return groupHas(g,truthNeedles.extral[state.truth.choices?.species])||groupHas(g,truthNeedles.extral[state.truth.choices?.network]);
  }
  return true;
}

function truthPrereqMatches(t,available=truthAvailableTalents()){
  if(!t.prerequisiteName)return [];
  const p=normText(t.prerequisiteName);
  return available.filter(x=>x.id!==t.id&&p.includes(normText(x.name))).map(x=>x.id);
}
function truthPrereqSatisfied(t,available=truthAvailableTalents()){
  const req=truthPrereqMatches(t,available);
  return !req.length||req.every(id=>state.truth.truthTalents.includes(id));
}

function economic(){
  const st=style(); if(!st)return {account:0,envelope:0,vehicleCapital:0,gen2:0};
  const envelope=st.augmentationEnvelope+state.edge.augmentationPacks*ruleset.creation.edge.augmentationPack;
  const vehicleCapital=st.vehicleCapital||0;
  const augOverflow=Math.max(0,state.spending.augmentations-envelope);
  const augUnused=Math.max(0,envelope-state.spending.augmentations);
  const vehOverflow=Math.max(0,state.spending.vehicle-vehicleCapital);
  const vehUnused=Math.max(0,vehicleCapital-state.spending.vehicle);
  const account=st.account + state.edge.cashPacks*ruleset.creation.edge.cashPack - state.spending.equipment - augOverflow - vehOverflow + augUnused*.5 + vehUnused*.5;
  return {account,envelope,vehicleCapital,gen2:(st.gen2SlotsBase||0)+state.edge.augmentationPacks,augUnused,vehUnused,augOverflow,vehOverflow};
}
