function t49ChoiceLabel(key){
  const c=truthNature()?.choices?.find(x=>x.key===key),id=state.truth.choices?.[key];
  return c?.options?.find(x=>x.id===id)?.name||'';
}
function t49StatsForNature(){
  const n=state.truth.nature,c=state.truth.choices||{};
  if(n==='humain')return {label:'Humain',v:'Aucun bonus de Nature',sr:'Aucun bonus racial : être Initié n’est pas une transformation physique',r:'Aucun bonus racial : un Humain ne possède pas de forme Révélée propre'};
  if(n==='vampire')return {label:'Nature Vampire',v:'Aucun bonus de Nature',sr:'+1 Vigueur · +1 Volonté',r:'+2 Vigueur · +1 Agilité · +1 Volonté'};
  if(n==='mage')return {label:'Mageius ouvert',v:'Aucune modification',sr:'+1 Esprit · +1 Volonté',r:'+1 Esprit · +2 Volonté'};
  if(n==='daemon'){const x=t46DaemonRevealStats[c.divinity];return {label:t49ChoiceLabel('divinity')||'Divinité à choisir',v:'Aucun bonus daemoniaque d’Attribut',sr:x?.sr||'Choisir une Divinité',r:x?.r||'Choisir une Divinité'};}
  if(n==='angelus'){const x=t49AngelusStats[c.sephirah];return {label:t49ChoiceLabel('sephirah')||'Sephirah à choisir',v:'Aucun bonus céleste d’Attribut',sr:x?.sr||'Choisir une Sephirah',r:x?.r||'Choisir une Sephirah'};}
  if(n==='aseryn'){const x=t49AserynStats[c.origin];return {label:t49ChoiceLabel('origin')||'Origine aseryne à choisir',v:'Aucun modificateur racial commun',sr:x?.sr||'+1 Agilité commun, puis empreinte de l’Origine',r:x?.r||'+2 Agilité commun, puis empreinte de l’Origine'};}
  if(n==='exile'){const x=t49ExileStats[c.people];return {label:t49ChoiceLabel('people')||'Peuple exilé à choisir',v:'Traduction humaine/plausible · aucun bonus racial',sr:x?.sr||'Choisir un peuple',r:x?.r||'Choisir un peuple'};}
  if(n==='extral'){const x=t49ExtralStats[c.species];return {label:t49ChoiceLabel('species')||'Profil extral à choisir',v:'Traduction humaine/plausible · aucun bonus racial',sr:x?.sr||'Choisir un profil',r:x?.r||'Choisir un profil'};}
  if(n==='garou'){
    if(c.blood==='sang_enchaine')return {label:'Garou · Sang Enchaîné',v:'Aucun bonus de Nature',sr:'Aucun bonus d’Attribut automatique · instincts/sens SR',r:'+2 Vigueur · +2 Agilité · +1 PA/round · forme hybride interdite'};
    return {label:'Garou',v:'Aucun bonus de Nature',sr:'Aucun bonus d’Attribut automatique · instincts/sens SR',r:'Humain révélé : aucun gros bonus automatique · Loup : +2 Agilité · Hybride : +3 Vigueur · +2 Agilité · +1 PA/round'};
  }
  if(n==='khinae'){
    const base=t49KhinaeBase[c.lineage],variant=t49KhinaeVariant[c.lineage]?.[c.variant],suffix=variant?` · Variante hybride sélectionnée : ${variant}`:'';
    return {label:[t49ChoiceLabel('lineage'),t49ChoiceLabel('variant')].filter(Boolean).join(' · ')||'Lignée à choisir',v:'Aucun bonus de Nature',sr:'Aucun bonus d’Attribut commun · instincts/perceptions SR de Lignée',r:base?`Humain révélé : aucun bonus d’Attribut automatique · Animal : ${base.animal} · Hybride : ${base.hybrid}${suffix}`:'Choisir une Lignée'};
  }
  return {label:truthNature()?.name||'Nature',v:'Aucun bonus chiffré renseigné',sr:'Voir capacités SR',r:'Voir capacités R'};
}
