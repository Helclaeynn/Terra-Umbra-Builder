function r18Money(v){if(v===null||v===undefined)return 0;if(typeof v==='number')return Number.isFinite(v)?v:0;const m=String(v).replace(/\s/g,'').match(/\d+(?:[.,]\d+)?/);return m?Number(m[0].replace(',','.')):0}
function r18AugMaintenance(){
  const value=state.reality.augmentations.reduce((s,p)=>s+lockPurchaseCost(p),0);if(value<=0)return 0;
  const levels=[25,100,250,500,1000,1500];let idx=value<=5000?0:value<=20000?1:value<=50000?2:value<=100000?3:value<=200000?4:5;
  const majors=state.reality.augmentations.map(p=>purchaseItem(p)).filter(x=>x&&x.generation===1&&r15IsSupport(x)&&/cablage neuronal|cyberoeil|cyberaudio|cyberbras|cyberjambe|support/.test(normText(x.name||''))).length;
  if(majors>=2)idx=Math.min(levels.length-1,idx+1);return levels[idx];
}
