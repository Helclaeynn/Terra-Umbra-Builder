function p40Field(label,help,value,oninput,placeholder=''){

function p40ValidationReason(id,ok){
  if(ok)return 'Validé';
  const map={
    identity:'Nom et âge requis.',origin:'Origine et Talent d’Origine requis.',sphere:'Sphère, Style et 5 points guidés requis.',attributes:'Budget d’Attributs à terminer.',skills:'Répartition des Compétences à terminer.',talents:'Choix de Talents ou prérequis à corriger.',truth:'Choix de Vérité incomplet ou étape vide non confirmée.',disadvantages:'Désavantages à corriger ou étape vide non confirmée.',edge:'Dépenses/attributions Edge à corriger ou étape vide non confirmée.',equipment:'Équipement à corriger ou étape vide non confirmée.',finish:'Finalisation sociale ou validation globale incomplète.'
  };
  return map[id]||'À compléter';
}
function p40ValidationPanel(){
  const v=validations(),box=el('section',{class:'p40-validation-panel'},'<div class="section-title reality-subtitle"><div><h3>Contrôle de validation</h3><p>Chaque voyant reprend exactement la logique utilisée par le builder pour déclarer la fiche valide.</p></div></div>'),grid=el('div',{class:'p40-validation-grid'});
  STEPS.forEach(([id,label],i)=>{const ok=!!v[id],row=el('button',{class:`p40-validation-switch ${ok?'ok':'bad'}`,type:'button'},`<span class="p40-switch-dot"></span><span><strong>${i+1}. ${uxEsc(label)}</strong><small>${uxEsc(p40ValidationReason(id,ok))}</small></span>`);row.onclick=()=>{stepIndex=i;render()};grid.appendChild(row)});
  box.appendChild(grid);return box;
}
