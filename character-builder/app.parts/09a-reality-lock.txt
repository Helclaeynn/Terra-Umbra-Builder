function lockFinite(v){if(v===null||v===undefined||v==='')return null;const n=Number(v);return Number.isFinite(n)?n:null}
function lockRaw(obj,key){return obj&&Object.prototype.hasOwnProperty.call(obj,key)?obj[key]:null}
function lockPriceSpec(item){
  const raw=item?.raw||{};
  let mode=normText(String(lockRaw(raw,'priceMode')||lockRaw(raw,'price_mode')||''));
  const label=String(lockRaw(raw,'priceLabel')||lockRaw(raw,'price_label')||realityField(raw,['Prix','Prix de reference','Prix indicatif','price','cout','coût'])|| (item?.price!==null&&item?.price!==undefined?fmt(item.price):'Prix à définir')).trim();
  let exact=lockFinite(lockRaw(raw,'price'));
  if(exact===null&&mode==='exact')exact=lockFinite(item?.price);
  let min=lockFinite(lockRaw(raw,'priceMin'));if(min===null)min=lockFinite(lockRaw(raw,'price_min'));
  let max=lockFinite(lockRaw(raw,'priceMax'));if(max===null)max=lockFinite(lockRaw(raw,'price_max'));
  if(!mode){
    const n=normText(label);
    if(/\bx\s*\d/.test(n)||n.includes('prix normal'))mode='multiplier';
    else if(/\d\s*[-–]\s*\d/.test(label))mode='range';
    else if(/\$\s*\+|\+\s*$/.test(label))mode='minimum';
    else if(n.includes('inclus')||n.includes('gratuit'))mode='included';
    else if(lockFinite(item?.price)!==null)mode='exact';
    else mode='manual';
  }
  if(['fixed','reference','prix de reference'].includes(mode))mode='exact';
  if(['min','minimum plus','at least'].includes(mode))mode='minimum';
  if(['variable','custom','unknown','indicatif'].includes(mode))mode='manual';
  if(['free','gratuit','included','inclus'].includes(mode))mode='included';
  if(mode==='exact'&&exact===null)exact=lockFinite(item?.price);
  if(mode==='range'){
    if(min===null)min=lockFinite(item?.price);
    if(max===null)max=min;
  }
  if(mode==='minimum'&&min===null)min=lockFinite(item?.price);
  if(mode==='included'){exact=0;min=0;max=0}
  const configurable=['range','minimum','multiplier','manual'].includes(mode);
  const defaultCost=mode==='exact'?exact:mode==='included'?0:(mode==='range'||mode==='minimum')?min:null;
  return {mode,label,exact,min,max,configurable,defaultCost};
}

    const mode=normText(String(row.priceMode||''));

      vehicle:sourceType==='vehicle',neuro:sourceType==='neuroprogram',sourceType,raw:row,details:row.data||null
    };
  });
}

function lockPurchaseCost(p){
  const item=purchaseItem(p);if(!item)return 0;const spec=lockPriceSpec(item);
  const selected=lockFinite(p.selectedPrice);if(selected!==null)return selected;
  return Number.isFinite(Number(spec.defaultCost))?Number(spec.defaultCost):0;
}
function lockPriceConfigured(p){
  const item=purchaseItem(p);if(!item)return false;const spec=lockPriceSpec(item),cost=lockPurchaseCost(p);
  if(spec.mode==='exact'||spec.mode==='included')return true;
  if(!p.priceConfirmed)return false;
  if(spec.mode==='range')return spec.min!==null&&spec.max!==null&&cost>=spec.min&&cost<=spec.max;
  if(spec.mode==='minimum')return spec.min!==null&&cost>=spec.min;
  if(['manual','multiplier'].includes(spec.mode))return Number.isFinite(cost)&&cost>0;
  return false;
}
function lockNormalizePurchasePrice(p){
  const item=purchaseItem(p);if(!item)return;const spec=lockPriceSpec(item);
  if(spec.mode==='exact'||spec.mode==='included'){p.selectedPrice=spec.defaultCost??0;p.priceConfirmed=true;return}
  if((spec.mode==='range'||spec.mode==='minimum')&&lockFinite(p.selectedPrice)===null)p.selectedPrice=spec.defaultCost??0;
  if(!Object.prototype.hasOwnProperty.call(p,'priceConfirmed'))p.priceConfirmed=false;
}
