// TUC Web V2 bridge — activates only for /builder/?character=<uuid>.
// In V2 mode the PostgreSQL character JSON is the source of truth; localStorage
// remains available only to the standalone V1 Builder.

const v2CharacterId=String(window.__TUC_V2_CHARACTER_ID__||'');
if(v2CharacterId){
  const v2StatusHost=$('#globalAlerts');
  const v2Status=el('div',{id:'v2CharacterSyncStatus',class:'v2-character-sync'},'Chargement de la fiche serveur…');
  v2StatusHost?.appendChild(v2Status);

  const v2Css=el('style',{},`
    .v2-character-sync{margin:0 0 12px;padding:10px 12px;border:1px solid #40505e;border-radius:10px;background:#0b141b;color:var(--muted);font-size:12px;line-height:1.45}
    .v2-character-sync strong{color:var(--text)}
    .v2-character-sync.ok{border-color:#365f45;color:#a8cdb0}
    .v2-character-sync.error{border-color:#7a3d3d;color:#e2a5a5}
  `);
  document.head.appendChild(v2Css);

  const setV2Status=(message,kind='')=>{
    if(!v2Status)return;
    v2Status.className=`v2-character-sync ${kind}`.trim();
    v2Status.innerHTML=message;
  };

  const fetchCharacter=async()=>{
    if(window.__TUC_V2_BOOTSTRAP__)return await window.__TUC_V2_BOOTSTRAP__;
    const response=await fetch(`/api/characters/${encodeURIComponent(v2CharacterId)}`,{credentials:'same-origin',cache:'no-store'});
    const body=await response.json().catch(()=>({}));
    if(!response.ok)throw Object.assign(new Error(body.error||`http_${response.status}`),{status:response.status});
    return body;
  };

  let v2Character;
  try{
    const loaded=await fetchCharacter();
    v2Character=loaded?.character;
    if(!v2Character||v2Character.id!==v2CharacterId)throw new Error('character_not_found');

    const serverData=v2Character.data&&typeof v2Character.data==='object'&&!Array.isArray(v2Character.data)
      ? structuredClone(v2Character.data)
      : {};

    // All localStorage restores are disabled in V2 mode before this point, so an
    // empty server object intentionally means a fresh Builder state.
    if(Object.keys(serverData).length)mergeState(serverData);
    if(!String(state.identity?.name||'').trim())state.identity.name=v2Character.name;
    render();

    const saveBtn=$('#saveBtn');
    if(saveBtn){
      saveBtn.textContent='Enregistrer';
      saveBtn.title='Enregistrer cette fiche dans votre compte Terra Umbra';
    }

    const newCharacterBtn=$('#newCharacterBtn');
    if(newCharacterBtn){
      newCharacterBtn.textContent='Mes personnages';
      newCharacterBtn.title='Retourner à la liste de vos personnages';
      newCharacterBtn.onclick=()=>{location.href='/'};
    }

    async function saveV2Character(){
      if(!v2Character)return null;
      const button=$('#saveBtn');
      if(button)button.disabled=true;
      setV2Status(`<strong>${v2Character.name}</strong> · enregistrement en cours…`);
      try{
        const name=String(state.identity?.name||v2Character.name||'').trim()||v2Character.name;
        const payload={
          name,
          data:structuredClone(state),
          version:v2Character.version
        };
        const response=await fetch(`/api/characters/${encodeURIComponent(v2CharacterId)}`,{
          method:'PATCH',
          credentials:'same-origin',
          cache:'no-store',
          headers:{'Content-Type':'application/json'},
          body:JSON.stringify(payload)
        });
        const body=await response.json().catch(()=>({}));
        if(response.status===409){
          setV2Status('<strong>Conflit de version.</strong> Cette fiche a été modifiée ailleurs. Rechargez la page pour récupérer la version la plus récente avant de continuer.','error');
          return null;
        }
        if(!response.ok)throw Object.assign(new Error(body.error||`http_${response.status}`),{status:response.status});
        v2Character=body.character;
        setV2Status(`<strong>${v2Character.name}</strong> · sauvegardé sur le serveur · version ${v2Character.version}`,'ok');
        if(button){
          button.textContent='Enregistré ✓';
          setTimeout(()=>{if(button)button.textContent='Enregistrer'},1000);
        }
        return structuredClone(v2Character);
      }catch(error){
        console.error('TUC V2 character save',error);
        const auth=error?.status===401?' La session a expiré : reconnectez-vous depuis Mes personnages.':'';
        setV2Status(`<strong>Échec de la sauvegarde serveur.</strong>${auth}`,'error');
        return null;
      }finally{
        if(button)button.disabled=false;
      }
    }

    if(saveBtn)saveBtn.onclick=saveV2Character;
    setV2Status(`<strong>${v2Character.name}</strong> · chargé depuis le serveur · version ${v2Character.version}`,'ok');

    window.TUCV2Bridge={
      id:v2CharacterId,
      get version(){return v2Character?.version||0},
      snapshot:()=>structuredClone(state),
      save:saveV2Character
    };
  }catch(error){
    console.error('TUC V2 character load',error);
    const auth=error?.status===401?' Votre session est absente ou expirée.':'';
    setV2Status(`<strong>Impossible de charger cette fiche.</strong>${auth} Retournez à Mes personnages puis réessayez.`,'error');
    throw error;
  }
}
