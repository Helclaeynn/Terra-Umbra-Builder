function r37Sentence(v=''){
  let s=String(v||'').trim().replace(/\s+/g,' ');if(!s)return '';
  s=s.charAt(0).toLowerCase()+s.slice(1);return /[.!?]$/.test(s)?s:`${s}.`;
}

function r37ItemLore(item){

function r37RecurringKind(item){
  if(!item)return 'one_off';
  const n=normText(item.name||''),key=r34Loose(item.name||''),cat=normText(`${item.category||''} ${r37DeepText(item,['Type','Section','Famille'])}`),price=normText(r37PriceText(item));
  const context=normText(`${n} ${cat} ${r37DeepText(item,['Fréquence','Frequence','Périodicité','Periodicite','Facturation'])}`);

  // A vehicle line is the purchase of the vehicle. Its Entretien/mois field is handled automatically elsewhere.
  if(item.vehicle)return 'durable_purchase';

  // Explicit per-use/session billing must win over broad service/category matches.
  if(/par trajet|\/\s*trajet|par intervention|\/\s*intervention|par seance|\/\s*seance|par heure|\/\s*h\b|par jour|\/\s*jour|par verre|\/\s*verre|par tasse|\/\s*tasse/.test(price)||key==='metro tram'||/^mas (court trajet|trajet long urbain)$/.test(key))return 'per_use';

  if(/par an|\/\s*an\b|annuel|annuelle|annuels|annuelles/.test(price)||/abonnement annuel/.test(context))return 'annual';
  if(/par mois|\/\s*mois\b|mensuel|mensuelle|mensuels|mensuelles/.test(price)||/abonnement mensuel|dette mensuelle|pension mensuelle|loyer|leasing/.test(context))return 'monthly';

  // V9 treats housing and hideouts as durable monthly commitments even when a zero-cost row omits “/mois”.
  if(/logement|planque/.test(cat)||/dortoir ouvrier|vladic micro logement|vladic studio|studio urbain ancien|appartement connecte|vladic familial|vladic grand|residence de cadre|penthouse|villa/.test(n)||/cache improvisee|piece box squat discret|studio garage anonyme|planque dediee|planque securisee|reseau de planques/.test(n))return 'monthly';

  // Canonical mobility and insurance subscriptions whose recurrence is semantic as well as textual.
  if(key==='pass metro tram'||/^bull (basic|standard|premium|executive)$/.test(key))return 'monthly';
  if(/^careforce (bronze|silver|golden)$/.test(key))return 'annual';
  if(/stockage box atelier|media holonet reseaux/.test(key))return 'monthly';

  if(/abonnement|assurance|careforce/.test(context))return price.includes('an')?'annual':price.includes('mois')?'monthly':'one_off';
  if(/vehicule|arme|armure|augmentation|materiel|objet/.test(cat))return 'durable_purchase';
  return 'one_off';
}
function r37RecurringCatalogItem(item){const k=r37RecurringKind(item);return k==='monthly'||k==='annual'}
r15RecurringCatalogItem=function(item){return r37RecurringCatalogItem(item)};

function r37MonthlyAmount(item){
  const kind=r37RecurringKind(item),cost=Math.max(0,Number(item?.price)||0);
  return kind==='annual'?cost/12:kind==='monthly'?cost:0;
}
function r37RecurringGroup(item){
  const n=normText(item?.name||''),cat=normText(item?.category||'');
  if(/planque/.test(cat)||/cache improvisee|piece box squat discret|studio garage anonyme|planque/.test(n))return 'Planque';
  if(/logement/.test(cat)||/dortoir|micro logement|vladic studio|studio urbain|appartement|residence|penthouse|villa/.test(n))return 'Logement';
  if(/transport|mobilite/.test(cat)||/^pass metro tram$/.test(n)||/^bull (basic|standard|premium|executive)$/.test(n))return 'Mobilité';
  if(/careforce|assurance|soin|sante/.test(`${n} ${cat}`))return 'Assurance & soins';
  if(/application|service|loisir|holonet|stockage|abonnement/.test(`${n} ${cat}`))return 'Abonnements & services';
  return 'Autres engagements';
}
function r37RecurringItems(){
  return realityData.equipment.filter(r37RecurringCatalogItem).sort((a,b)=>r37RecurringGroup(a).localeCompare(r37RecurringGroup(b),'fr')||a.name.localeCompare(b.name,'fr'));
}
r18RecurringItems=function(){return r37RecurringItems()};
r18RecurringGroups=function(){
  const order=['Logement','Planque','Mobilité','Assurance & soins','Abonnements & services','Autres engagements'],map=new Map(order.map(x=>[x,[]]));
  for(const item of r37RecurringItems()){const g=r37RecurringGroup(item);if(!map.has(g))map.set(g,[]);map.get(g).push(item)}
  return [...map.entries()].filter(([,items])=>items.length).map(([label,items])=>({label,items}));
};

function r37NormalizeLegacyRecurring(){
  r15EnsureCharges();let changed=false;
  for(const charge of state.reality.fixedChargeItems){
    if(!charge.sourceItemId)continue;
    const item=realityData.equipment.find(x=>x.id===charge.sourceItemId);if(!item||!r37RecurringCatalogItem(item))continue;
    const monthly=r37MonthlyAmount(item);if(Number(charge.monthly)!==monthly){charge.monthly=monthly;changed=true}
  }
  const keep=[];
  for(const purchase of state.reality.equipment){
    const item=purchaseItem(purchase);
    if(item&&r37RecurringCatalogItem(item)){
      if(!state.reality.fixedChargeItems.some(x=>x.sourceItemId===item.id))state.reality.fixedChargeItems.push({uid:`fc-${Date.now().toString(36)}-${Math.random().toString(36).slice(2,6)}`,name:item.name,monthly:r37MonthlyAmount(item),sourceItemId:item.id});
      changed=true;
    }else keep.push(purchase);
  }
  if(keep.length!==state.reality.equipment.length)state.reality.equipment=keep;
  if(changed&&typeof persist==='function')persist();
}

function r37BillingLabel(item){
  const kind=r37RecurringKind(item),price=r37PriceText(item),monthly=r37MonthlyAmount(item);
  if(kind==='annual')return `${price||fmt(item.price)+'/an'} · retenu ${fmt(monthly)}/mois`;
  if(kind==='monthly')return `${price||fmt(item.price)+'/mois'} · retenu ${fmt(monthly)}/mois`;
  if(kind==='per_use')return `${price||'coût par usage'} · dépense ponctuelle`;
  return `${price||lockPriceSpec(item).label} · achat ponctuel`;
}
