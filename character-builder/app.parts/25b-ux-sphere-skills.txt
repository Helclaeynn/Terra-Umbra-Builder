function p25SkillTotal(id){return typeof p20SkillFinal==='function'?p20SkillFinal(id):skillRaw(id)}
function p25ResetStylePoints(){Object.values(state.skills).forEach(v=>v.style=0)}
function p25SphereCards(){return Object.entries(ruleset.spheres).map(([id,s])=>({id,name:s.name,lore:p25SphereLore[id]||'',meta:`<strong>5 points fixes :</strong> ${s.fixedSkills.map(x=>`${uxEsc(skillName(x))} +1`).join(' · ')}<br><strong>Appui :</strong> ${uxEsc(s.support||'—')}`}))}
function p25StyleCards(){return ruleset.styles.filter(s=>s.sphere===state.creation.sphere).map(s=>({id:s.id,name:s.name,lore:p25StyleLore[s.id]||'',meta:`<strong>Train de vie :</strong> ${uxEsc(s.lifestyle)} · <strong>Compte :</strong> ${fmt(s.account)} · <strong>Aug. :</strong> ${fmt(s.augmentationEnvelope)}${s.vehicleCapital?` · <strong>Capital véhicule :</strong> ${fmt(s.vehicleCapital)}`:''}<br><strong>Compétences :</strong> ${s.skills.map(skillName).map(uxEsc).join(' · ')}<br><strong>Expertise :</strong> ${s.expertiseFamilies.map(attrName).map(uxEsc).join(' / ')}`}))}
function p25StyleAllocator(c){
  const st=style();if(!st)return;const used=totalStylePoints();
  c.appendChild(el('div',{class:'section-title reality-subtitle'},`<div><h3>Répartir les 5 points du Style</h3><p>Ces points représentent la pratique professionnelle ou opérationnelle de départ. Répartissez exactement 5 points entre les cinq Compétences du Style, avec un maximum de +2 dans une même Compétence.</p></div><div class="kpi ${used===5?'good':'bad'}">${used}/5</div>`));
  const grid=el('div',{class:'p25-style-allocator'});st.skills.forEach(id=>{const sk=ruleset.skills.find(x=>x.id===id),o=state.skills[id],total=p25SkillTotal(id),raw=skillRaw(id),box=el('div',{class:'alloc p25-skill-mini'},`<div class="alloc-title"><strong>${uxEsc(sk?.name||id)}</strong><span>Total ${total}</span></div><div class="p25-mini-help">${uxEsc(p25SkillHelp[id]||'')}</div><div class="stepper"><button data-dir="-1">−</button><strong>${o.style}</strong><button data-dir="1">+</button></div>`);box.querySelectorAll('button').forEach(b=>b.onclick=()=>{const dir=Number(b.dataset.dir),nv=Number(o.style||0)+dir;if(nv<0||nv>2)return;if(dir>0&&(totalStylePoints()>=5||raw>=5))return;o.style=nv;render()});grid.appendChild(box)});c.appendChild(grid)
}
stepSphere=function(){
  const c=wrap('Sphère actuelle & Style','Cette étape se lit comme un ensemble. La Sphère décrit le milieu, l’institution ou le réseau qui structure votre vie actuelle et pose immédiatement 5 points fixes de Compétence. Le Style décrit ce que vous savez faire au départ : il ajoute 5 points guidés à répartir ici, ouvre deux familles d’Expertise et fixe votre situation économique et augmentique avant les points libres.');
  c.appendChild(el('div',{class:'rulebox'},'<strong>Sphère ≠ Origine :</strong> l’Origine raconte d’où vous venez ; la Sphère raconte où vous vivez et travaillez aujourd’hui. <strong>Style ≠ classe :</strong> il décrit votre pratique de départ mais ne vous enferme pas dans une progression. L’Enveloppe augmentique indiquée par le Style n’est pas de l’argent liquide : elle représente ce qui a déjà pu être investi dans le corps avant le scénario 1.'));
  c.appendChild(p25ChoiceGrid(p25SphereCards(),state.creation.sphere,id=>{if(state.creation.sphere!==id){state.creation.sphere=id;state.creation.style='';state.talents.sphere='';state.talents.expertise='';p25ResetStylePoints()}render()}));
  if(!state.creation.sphere)return;
  const sp=sphere();c.appendChild(el('div',{class:'p25-fixed-reminder'},`<strong>Points de Sphère déjà posés :</strong> ${sp.fixedSkills.map(id=>`${uxEsc(skillName(id))} +1`).join(' · ')}`));
  c.appendChild(el('h3',{class:'subhead'},'Choisir le Style'));
  c.appendChild(p25ChoiceGrid(p25StyleCards(),state.creation.style,id=>{if(state.creation.style!==id){state.creation.style=id;state.talents.expertise='';p25ResetStylePoints()}render()}));
  if(style()){p25StyleAllocator(c);c.appendChild(el('div',{class:'rulebox'},`<strong>Appui de Sphère :</strong> ${uxEsc(sp.support||'—')}<br><strong>Expertise gratuite plus tard :</strong> une Expertise appartenant à ${style().expertiseFamilies.map(attrName).map(uxEsc).join(' ou ')}.`))}
};

stepSkills=function(){
  const st=style(),ok=totalFreePoints()===15;const c=wrap('Compétences libres','Les 5 points de Sphère et les 5 points de Style sont déjà intégrés à vos valeurs. Répartissez ici uniquement les 15 points libres. Le Total affiché est la valeur actuelle de la Compétence ; le plafond de création reste 5 avant les éventuels bonus explicitement accordés par un Talent.',`<div class="kpi ${ok?'good':'bad'}">Libres ${totalFreePoints()} / 15</div>`);
  if(!st){c.appendChild(el('div',{class:'empty'},'Choisissez d’abord une Sphère et un Style.'));return}
  if(totalStylePoints()!==5)c.appendChild(el('div',{class:'rulebox warn'},`Le Style n’est pas encore complètement réparti : ${totalStylePoints()}/5. Revenez au bloc Sphère & Style pour terminer ces points avant les points libres.`));
  ruleset.attributes.forEach(a=>{c.appendChild(el('h3',{class:'subhead'},a.name));const grid=el('div',{class:'alloc-grid p25-skill-grid'});ruleset.skills.filter(s=>s.attribute===a.id).forEach(sk=>{const o=state.skills[sk.id],raw=skillRaw(sk.id),total=p25SkillTotal(sk.id),box=el('div',{class:'alloc p25-skill-card'},`<div class="alloc-title"><strong>${uxEsc(sk.name)}</strong><span>Total ${total}</span></div><div class="p25-skill-help">${uxEsc(p25SkillHelp[sk.id]||'Compétence utilisée avec son Attribut de référence dans les situations correspondantes.')}</div>${total!==raw?'<div class="p25-bonus-note">Bonus permanent de Talent déjà inclus dans le Total.</div>':''}<div class="inline p25-free-line"><span class="tag">Points libres</span><div class="stepper"><button data-dir="-1">−</button><strong>${o.free}</strong><button data-dir="1">+</button></div></div>`);box.querySelectorAll('button').forEach(b=>b.onclick=()=>{const dir=Number(b.dataset.dir),nv=Number(o.free||0)+dir;if(nv<0)return;if(totalFreePoints()+dir<0||totalFreePoints()+dir>15)return;if(raw+dir>5)return;o.free=nv;render()});grid.appendChild(box)});c.appendChild(grid)});
};
