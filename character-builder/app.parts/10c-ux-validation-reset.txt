const uxBaseValidations=validations;
validations=function(){
  const v=uxBaseValidations(),st=style();
  v.attributes=attrTotal()===22&&uxAttrIds.every(id=>Number(state.attributes[id])>=3&&Number(state.attributes[id])<=7&&finalAttr(id)<=7);
  const styleOK=st&&totalStylePoints()===5&&Object.entries(state.skills).every(([id,o])=>o.style>=0&&o.style<=2&&(!o.style||st.skills.includes(id)));
  v.skills=!!styleOK&&totalFreePoints()===15&&ruleset.skills.every(s=>skillRaw(s.id)<=5);
  const expert=talentById(state.talents.expertise),expertFamilyOK=!expert||st?.expertiseFamilies.includes(expert.attribute),neuroOK=!hasTalent('neurodriver')||(skillRaw('neurodive')>=1&&!hasDis('unsinkable'));
  v.talents=!!state.talents.origin&&!!state.talents.sphere&&!!state.talents.expertise&&!!state.talents.common&&expertFamilyOK&&neuroOK;
  const attrEdgeOK=!state.edge.attributePack||uxEdgeAttrTotal()===ruleset.creation.attributes.edgePackPoints;
  const skillEdgeOK=totalEdgeSkillPoints()===edgeSkillBudget(),talentEdgeOK=state.talents.edge.length===state.edge.talentPacks&&state.talents.edge.every(Boolean);
  v.edge=edgeRemaining()>=0&&state.edge.attributePack<=1&&state.edge.skillPacks<=3&&state.edge.talentPacks<=2&&state.edge.cashPacks<=3&&state.edge.lifestylePack<=1&&state.edge.augmentationPacks<=2&&state.edge.renownPack<=1&&attrEdgeOK&&skillEdgeOK&&talentEdgeOK;
  const langCount=1+skillRaw('langages_argot'),languagesOK=state.social.languages.length===langCount&&state.social.languages.every(x=>String(x).trim()),crawlerContact=state.creation.sphere!=='crawler'||state.social.contacts.some(x=>String(x).trim()),corpoSupport=state.creation.sphere!=='corporatiste'||!!state.reality.sphereSupportDetail.trim(),renownMilieuOK=renown()===0||!!String(state.social.renownMilieu||'').trim();
  v.finish=Object.entries(v).filter(([k])=>k!=='finish').every(([,ok])=>ok)&&languagesOK&&crawlerContact&&corpoSupport&&renownMilieuOK;
  return v;
};
const uxBaseCurrentErrors=currentErrors;

  let errs=uxBaseCurrentErrors().filter(x=>!x.startsWith('Attributs :')&&!x.startsWith('Compétences libres :')&&!x.startsWith('Compétences Edge :'));
  if(attrTotal()!==22)errs.push(`Attributs de création : ${attrTotal()} / 22 points.`);if(totalFreePoints()!==15)errs.push(`Compétences libres : ${totalFreePoints()} / 15 points.`);
  const here=STEPS[stepIndex]?.[0];if(here==='edge'||here==='finish'){if(state.edge.attributePack&&uxEdgeAttrTotal()!==2)errs.push(`Attributs Edge : ${uxEdgeAttrTotal()} / 2 points attribués.`);if(totalEdgeSkillPoints()!==edgeSkillBudget())errs.push(`Compétences Edge : ${totalEdgeSkillPoints()} / ${edgeSkillBudget()} points attribués.`);if(state.talents.edge.filter(Boolean).length!==state.edge.talentPacks)errs.push(`Talents Edge : ${state.talents.edge.filter(Boolean).length} / ${state.edge.talentPacks} choisi(s).`)}
  return [...new Set(errs)];
};
