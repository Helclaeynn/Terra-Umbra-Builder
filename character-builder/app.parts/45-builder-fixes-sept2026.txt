function p45MoneyNumbers(text){
  const matches=String(text||'').replace(/\u00a0/g,' ').match(/\d[\d\s]*(?:[.,]\d+)?/g)||[];
  return matches.map(x=>Number(x.replace(/\s/g,'').replace(',','.'))).filter(Number.isFinite);
}
function p45RecurringPriceInfo(item){
  const label=p45RecurringPriceText(item),nums=p45MoneyNumbers(label);let min=nums[0]??(Number.isFinite(Number(item?.price))?Number(item.price):0),max=nums[1]??min;
  const annual=/\/\s*(?:an|année|annee)|\bannuel|\bpar an\b/i.test(label);if(annual){min/=12;max/=12}
  min=Math.max(0,min||0);max=Math.max(min,max||min);const monthly=Math.round(((min+max)/2)/5)*5;
  return {label,minMonthly:min,maxMonthly:max,monthly,annual};
}
