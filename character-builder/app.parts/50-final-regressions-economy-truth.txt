const p50EnsureProgressionBase=p43EnsureProgression;
p43EnsureProgression=function(){
  const before=state.progression&&typeof state.progression==='object'?state.progression:{};
  const cashBase=before.cashBase;
  const cashTransactions=before.cashTransactions;
  p50EnsureProgressionBase();
  if(cashBase===null||Number.isFinite(Number(cashBase)))state.progression.cashBase=cashBase===null?null:Number(cashBase);
  if(Array.isArray(cashTransactions))state.progression.cashTransactions=cashTransactions;
  if(!Array.isArray(state.progression.cashTransactions))state.progression.cashTransactions=[];
};

function p50CommerceDegree(index){return p45CommerceDegrees[Math.max(0,Math.min(p45CommerceDegrees.length-1,Number(index)||0))]}
function p50PurchaseKind(purchase){
  if(purchase?.kind==='augmentation'||purchase?.kind==='equipment')return purchase.kind;
  if((state.reality.augmentations||[]).includes(purchase))return 'augmentation';
  return 'equipment';
}
p45CampaignPurchase=function(kind,item,listPrice,degreeIndex){
  const list=Math.max(0,Math.round(Number(listPrice)||0)),deg=p50CommerceDegree(degreeIndex),cost=Math.round(list*deg.buy);
  if(!item||!list)throw new Error('Prix catalogue à renseigner.');
  if(cost>p45Cash())throw new Error('Solde insuffisant.');
  if(kind==='augmentation'&&typeof r15SupportSatisfied==='function'&&!r15SupportSatisfied(item))throw new Error(`Support requis manquant : ${r15SupportLabel(item)}`);
  if(kind==='augmentation'&&item.generation===2&&gen2Windows()<1)throw new Error('Aucune fenêtre Gen2 disponible pour cette augmentation.');
  p45FreezeCashBase();
  const purchase={uid:`camp-${Date.now().toString(36)}-${Math.random().toString(36).slice(2,7)}`,itemId:item.id,kind,selectedPrice:cost,priceConfirmed:true,acquiredInCampaign:true,campaignCatalogPrice:list,campaignCommerceDegree:deg.id};
  if(kind==='augmentation'&&item.generation===2)purchase.gen2System=1;
  purchaseArray(kind).push(purchase);
  p45MoneyTxn(-cost,`Achat · ${item.name}`,'purchase');
  render();
};
p45CampaignSale=function(purchase,degreeIndex,kindOverride){
  const item=purchaseItem(purchase);if(!item)return;
  const kind=kindOverride||p50PurchaseKind(purchase),deg=p50CommerceDegree(degreeIndex),newPrice=Math.max(0,Number(purchase.campaignCatalogPrice??lockPurchaseCost(purchase))||0),sale=Math.round(newPrice*deg.sale);
  p45FreezeCashBase();
  const arr=purchaseArray(kind),i=arr.findIndex(x=>x===purchase||(purchase.uid&&x.uid===purchase.uid));if(i<0)return;
  arr.splice(i,1);p45MoneyTxn(sale,`Revente · ${item.name}`,'sale');render();
};

function p50TradePreview(item,base,deg,mode){
  if(!item)return '<div class="p50-trade-empty">Choisissez un article pour afficher le calcul et son effet.</div>';
  const factor=mode==='buy'?deg.buy:deg.sale,final=Math.round(base*factor),verb=mode==='buy'?'À payer':'À récupérer',commerce=deg.id===0?'Sans jet':`${deg.label} · ${deg.delta} %`;
  return `<div class="p50-trade-title"><strong>${uxEsc(item.name)}</strong><span>${mode==='buy'?'ACHAT':'REVENTE'}</span></div><div class="p50-trade-numbers"><div><small>Prix de référence</small><strong>${fmt(base)}</strong></div><div><small>Commerce</small><strong>${uxEsc(commerce)}</strong></div><div class="p50-trade-total"><small>${verb}</small><strong>${fmt(final)}</strong></div></div><p>${uxEsc(item.effect||r34ItemLore(item)||'')}</p>`;
}

}
