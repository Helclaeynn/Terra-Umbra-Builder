// Final Reality detail layer — augmentation supports, equipment profiles and fixed-charge Lifestyle economy.

const r15LifestyleRefs={Survie:200,Modeste:400,Standard:650,Confortable:1200,'Aisé':2500,Luxe:5000};
function r15DeepField(item,names){
  const pools=[item?.raw,item?.raw?.data,item?.details].filter(x=>x&&typeof x==='object');
  for(const p of pools){const v=realityField(p,names);if(v!==null&&v!==undefined&&v!=='')return v}return null;
}
function r15SupportRaw(item){return String(r15DeepField(item,['Support','Support requis','Support nécessaire','Support necessaire'])||'').trim()}
function r15PrereqRaw(item){return String(r15DeepField(item,['Prérequis','Prerequis','Prerequisite','Condition','Conditions'])||'').trim()}
function r15IsSupport(item){const n=normText(item?.name||''),t=normText(`${item?.category||''} ${r15DeepField(item,['Type'])||''}`);return t.includes('support')||/cablage neuronal|cyberoeil|cyberaudio|cyberbras|cyberjambe|cybermain|cyberpied|support de membres|support multi/.test(n)}
function r15SupportAlternatives(item){
  if(item?.kind!=='augmentation'||r15IsSupport(item))return [];
  const explicit=normText(r15SupportRaw(item)),n=normText(item.name||''),fam=augmentationFamilies(item),alts=[];
  const add=(...xs)=>{const a=xs.filter(Boolean);if(a.length)alts.push(a)};
  if(explicit){
    if(explicit.includes('cablage neuronal'))add('cablage neuronal');
    if(explicit.includes('cyberoeil'))add('cyberoeil');
    if(explicit.includes('cyberaudio'))add('cyberaudio');
    if(explicit.includes('cybermain')||explicit.includes('cyberbras'))add(...[explicit.includes('cybermain')?'cybermain':null,explicit.includes('cyberbras')?'cyberbras':null]);
    if(explicit.includes('cyberpied')||explicit.includes('cyberjambe'))add(...[explicit.includes('cyberpied')?'cyberpied':null,explicit.includes('cyberjambe')?'cyberjambe':null]);
  }
  if(!alts.length){
    if(fam.includes('neural'))add('cablage neuronal');
    else if(fam.includes('optical'))add('cyberoeil');
    else if(fam.includes('audio'))add('cyberaudio');
    else if(fam.includes('member')){
      if(/jambe|pied|palme/.test(n))add('cyberjambe','cyberpied');
      else add('cyberbras','cybermain');
    }
  }
  return alts;
}
function r15InstalledAugNames(){return state.reality.augmentations.map(p=>purchaseItem(p)).filter(Boolean).map(x=>normText(x.name||''))}
function r15SupportSatisfied(item){
  const reqs=r15SupportAlternatives(item);if(!reqs.length)return true;const installed=r15InstalledAugNames();
  return reqs.every(group=>group.some(needle=>installed.some(name=>name.includes(needle))));
}
function r15SupportLabel(item){
  const raw=r15SupportRaw(item);if(raw)return raw;
  const req=r15SupportAlternatives(item);if(!req.length)return '';
  return req.map(g=>g.map(x=>x.replace(/\b\w/g,c=>c.toUpperCase())).join(' ou ')).join(' + ');
}
const r15CanAffordBase=canAffordPurchase;
canAffordPurchase=function(kind,item){if(kind==='augmentation'&&!r15SupportSatisfied(item))return false;return r15CanAffordBase(kind,item)};
const r15ValidBase=validations;
validations=function(){const v=r15ValidBase();const supportOK=state.reality.augmentations.every(p=>{const x=purchaseItem(p);return !x||r15SupportSatisfied(x)});v.equipment=v.equipment&&supportOK;v.finish=Object.entries(v).filter(([k])=>k!=='finish').every(([,ok])=>ok);return v};
const r15ErrorsBase=currentErrors;
currentErrors=function(){const e=r15ErrorsBase(),bad=state.reality.augmentations.map(p=>purchaseItem(p)).filter(x=>x&&!r15SupportSatisfied(x));if(bad.length)e.push(`Supports augmentiques manquants : ${bad.map(x=>`${x.name} → ${r15SupportLabel(x)}`).join(', ')}.`);return [...new Set(e)]};

function r15EquipmentStats(item){
  const specs=[
    ['DGT',['DGT','Dégâts','Degats']],['Portée',['Portée','Portee']],['Capacité',['Capacité','Capacite','Coups','Chargeur']],['Propriétés',['Propriétés','Proprietes','Spéciaux','Speciaux']],
    ['Armure',['Armure','Valeur d’Armure','Valeur Armure']],['Balistique',['Balistique']],['Énergie',['Énergie','Energie']],['Protection',['Protection']],

function r15StatsBar(item){const stats=r15EquipmentStats(item);if(!stats.length)return null;return el('div',{class:'r15-statbar'},stats.map(([k,v])=>`<span><strong>${uxEsc(k)}</strong> ${uxEsc(v)}</span>`).join(''))}
