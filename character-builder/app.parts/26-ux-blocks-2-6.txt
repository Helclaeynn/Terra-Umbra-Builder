function p26SphereTalentFlavour(t){
  const id=normText(t?.id||''),sp=sphere()?.name||'votre organisation';
  if(/dotation|materiel|equipement/.test(id))return `Votre fonction implique que ${sp} vous confie réellement les outils nécessaires pour travailler.`;
  if(/assurance|couverture|juridique/.test(id))return `Votre appartenance ne disparaît pas quand un problème survient : une structure est censée vous couvrir dans son domaine.`;
  if(/badge|habilitation|registre|acces/.test(id))return `Votre fonction ouvre des portes, fichiers ou circuits qui restent normalement fermés à un simple extérieur.`;
  if(/reseau|contact/.test(id))return `Votre place dans ${sp} vous a donné des relations concrètes au-delà de votre équipe immédiate.`;
  if(/procedure|priorite|requisition/.test(id))return `Vous savez faire fonctionner les procédures de ${sp} suffisamment bien pour obtenir rapidement ce qu’une mission justifie.`;
  if(/appui|extraction|renfort/.test(id))return `Quand une situation professionnelle devient sérieuse, vous savez à qui demander de véritables moyens plutôt que de rester seul.`;
  if(/profil|dossier/.test(id))return `Votre identité professionnelle est assez cohérente et reconnue pour résister à un examen ordinaire de votre position.`;
  if(/compte|avantage|fournisseur|service/.test(id))return `Votre appartenance à ${sp} se traduit aussi par des avantages matériels ou des services auxquels tout le monde n’a pas accès.`;
  if(/autorite/.test(id))return `Dans le cadre où votre fonction est reconnue, votre parole porte le poids réel de l’institution ou du milieu derrière vous.`;
  return `Ce Talent représente un avantage concret acquis par votre place dans ${sp} : il existe dans la fiction avant d’exister dans les chiffres.`;
}
function p26TalentNarrative(t){
  if(!t)return '';
  return p25OriginTalentFlavour?.[t.id]||p26TalentFlavour[t.id]||(t.category==='sphere'?p26SphereTalentFlavour(t):`« ${t.name} » décrit une capacité ou un acquis réellement visible dans la manière dont votre personnage agit et vit.`);
}
function p26TalentDetail(t){
  if(!t)return null;
  return el('div',{class:'p26-talent-detail'},`<strong>${uxEsc(t.name)}</strong><em>${uxEsc(p26TalentNarrative(t))}</em><p><strong>Effet mécanique :</strong> ${uxEsc(t.effect||'—')}</p>`);
}
function p26TalentPickGrid(items,value,onSelect,{gridClass='p26-talent-pick-grid'}={}){
  const grid=el('div',{class:gridClass});
  items.forEach(t=>{
    const selected=value===t.id,card=el('label',{class:`p26-talent-pick ${selected?'selected':''}`}),radio=el('input',{type:'radio',checked:selected});
    radio.onchange=()=>onSelect(t.id);
    const main=el('span',{class:'p26-talent-pick-main'},`<strong>${uxEsc(t.name)}</strong>`);
    if(!selected)main.appendChild(uxInfo(t.effect||'Aucun effet mécanique renseigné.'));
    card.append(radio,main);grid.appendChild(card);
  });
  return grid;
}

function p26StyleAllocator(c){
  const st=style();if(!st)return;const used=totalStylePoints();
  c.appendChild(el('div',{class:'section-title reality-subtitle'},`<div><h3>Répartir les 5 points du Style</h3><p>Ces points représentent votre pratique professionnelle ou opérationnelle de départ. Répartissez exactement 5 points entre les cinq Compétences du Style, avec un maximum de +2 dans une même Compétence.</p></div><div class="kpi ${used===5?'good':'bad'}">${used}/5</div>`));
  const grid=el('div',{class:'p26-five-grid'});
  st.skills.forEach(id=>{
    const sk=ruleset.skills.find(x=>x.id===id),o=state.skills[id],total=p25SkillTotal(id),raw=skillRaw(id),box=el('div',{class:'alloc p26-skill-card'},`<div class="alloc-title"><strong>${uxEsc(sk?.name||id)}</strong><span>${total}/5</span></div><div class="p26-card-help">${uxEsc(p25SkillHelp[id]||'')}</div><div class="stepper"><button data-dir="-1">−</button><strong>${o.style}</strong><button data-dir="1">+</button></div>`);
    box.querySelectorAll('button').forEach(b=>b.onclick=()=>{const dir=Number(b.dataset.dir),nv=Number(o.style||0)+dir;if(nv<0||nv>2)return;if(dir>0&&(totalStylePoints()>=5||raw>=5))return;o.style=nv;render()});grid.appendChild(box);
  });c.appendChild(grid);
}

function p26SkillGroup(a,index){
  const owned=ruleset.skills.filter(s=>s.attribute===a.id).some(s=>Number(state.skills[s.id]?.free||0)>0),d=el('details',{class:'p26-skill-group',open:index===0||owned});
  d.appendChild(el('summary',{},`<span><strong>${uxEsc(a.name)}</strong><small>5 Compétences</small></span><span class="badge">ouvrir / replier</span>`));
  const grid=el('div',{class:'p26-five-grid p26-skill-group-grid'});
  ruleset.skills.filter(s=>s.attribute===a.id).forEach(sk=>{
    const o=state.skills[sk.id],total=p25SkillTotal(sk.id),box=el('div',{class:'alloc p26-skill-card'},`<div class="alloc-title"><strong>${uxEsc(sk.name)}</strong><span>${total}/5</span></div><div class="p26-card-help">${uxEsc(p25SkillHelp[sk.id]||'')}</div><div class="inline p26-skill-step"><span class="tag">Points libres</span><div class="stepper"><button data-dir="-1">−</button><strong>${o.free}</strong><button data-dir="1">+</button></div></div>`);
    box.querySelectorAll('button').forEach(b=>b.onclick=()=>{const dir=Number(b.dataset.dir),nv=Number(o.free||0)+dir;if(nv<0)return;if(totalFreePoints()+dir<0||totalFreePoints()+dir>15)return;if(skillRaw(sk.id)+dir>5)return;o.free=nv;render()});grid.appendChild(box);
  });d.appendChild(grid);return d;
}

function p26TalentSection(title,subtitle,items,value,onSelect,name,open=true){
  const d=el('details',{class:'ux-group p26-talent-section',open});
  d.appendChild(el('summary',{},`<span><strong>${uxEsc(title)}</strong>${subtitle?`<small>${uxEsc(subtitle)}</small>`:''}</span><span class="badge">${items.length}</span>`));
  d.appendChild(p26TalentPickGrid(items,value,onSelect));
  const chosen=items.find(x=>x.id===value);if(chosen)d.appendChild(p26TalentDetail(chosen));return d;
}
