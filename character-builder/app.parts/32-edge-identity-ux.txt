  p25EnsureIdentity();

  const layout=el('div',{class:'p25-identity-layout'}),portrait=el('aside',{class:'p25-portrait-card'}),frame=el('div',{class:`p25-portrait-frame ${state.identity.portraitDataUrl?'has-image':''}`});

    [p25LiveField('alias','Alias','Surnom, indicatif, nom de scène ou pseudonyme utilisé dans certains milieux.',{placeholder:'Ex. Ghost, Dr. Vale…'}),p25LiveField('sex','Sexe / genre','Information descriptive sans conséquence mécanique.',{placeholder:'Ex. femme, homme, non-binaire…'})],

  renownPack:{lore:'Le personnage arrive déjà avec un nom qui circule dans un milieu précis : réputation professionnelle, exploits, œuvre publique ou notoriété locale.',mechanic:'1 Edge · +1 Renommée de départ · maximum 1 fois · Renommée initiale maximum 2.'}

function p32EdgeSpendCard(label,key,max,extra='',canMinus=()=>true,canPlus=()=>true){
  const val=Number(state.edge[key]||0),txt=p32EdgeSpend[key]||{lore:'',mechanic:''},card=el('div',{class:'alloc edge-card p32-edge-spend'},`<div class="alloc-title"><strong>${uxEsc(label)}</strong><span>${val}/${max}</span></div><em>${uxEsc(txt.lore)}</em><p>${uxEsc(txt.mechanic)}</p>${extra?`<div class="edge-result">${extra}</div>`:''}<div class="stepper"><button data-dir="-1">−</button><strong>${val}</strong><button data-dir="1">+</button></div>`);
  const [minus,plus]=card.querySelectorAll('button');minus.disabled=val<=0||!canMinus();plus.disabled=val>=max||edgeRemaining()<=0||!canPlus();
  minus.onclick=()=>{if(val<=0||!canMinus())return;state.edge[key]=val-1;if(key==='talentPacks')trimEdgeTalentSlots();render()};
  plus.onclick=()=>{if(val>=max||edgeRemaining()<=0||!canPlus())return;state.edge[key]=val+1;if(key==='talentPacks')trimEdgeTalentSlots();render()};return card;
}

renderEdgeAttributeAllocator=function(c){
  uxEnsureEdgeAttributes();if(!state.edge.attributePack)return;const budget=ruleset.creation.attributes.edgePackPoints,used=uxEdgeAttrTotal();
  c.appendChild(el('div',{class:'section-title reality-subtitle'},`<div><h3>Attribuer les +2 Attributs</h3><p>Répartissez immédiatement les deux points achetés. Ils s’ajoutent aux valeurs de création sans modifier le budget de 22 points.</p></div><div class="kpi ${used===budget?'good':'bad'}">${used}/${budget}</div>`));
  const grid=el('div',{class:'p32-five-grid'});ruleset.attributes.forEach(a=>{const edge=Number(state.edgeAttributes[a.id]||0),fin=finalAttr(a.id),box=el('div',{class:'alloc p32-edge-attr'},`<div class="alloc-title"><strong>${uxEsc(a.name)}</strong><span>${fin}/7</span></div><div class="p26-card-help">${uxEsc(p26AttributeHelp[a.id]||'')}</div><div class="stepper"><button data-dir="-1">−</button><strong>+${edge}</strong><button data-dir="1">+</button></div>`);box.querySelectorAll('button').forEach(b=>b.onclick=()=>{const dir=Number(b.dataset.dir),nv=edge+dir;if(nv<0)return;if(dir>0&&(uxEdgeAttrTotal()>=budget||finalAttr(a.id)>=7))return;state.edgeAttributes[a.id]=nv;render()});grid.appendChild(box)});c.appendChild(grid);
};

renderEdgeSkillAllocator=function(c){
  if(!state.edge.skillPacks)return;const budget=edgeSkillBudget(),used=totalEdgeSkillPoints();
  c.appendChild(el('div',{class:'section-title reality-subtitle'},`<div><h3>Attribuer les points de Compétence Edge</h3><p>${state.edge.skillPacks} achat(s) donnent ${budget} points. Répartissez-les ici sans dépasser 5 dans une Compétence.</p></div><div class="kpi ${used===budget?'good':'bad'}">${used}/${budget}</div>`));
  ruleset.attributes.forEach(a=>{const eligible=ruleset.skills.filter(s=>s.attribute===a.id&&(state.skills[s.id].edge>0||skillRaw(s.id)<5));if(!eligible.length)return;const selected=eligible.some(s=>state.skills[s.id].edge>0),d=el('details',{class:'ux-group p32-edge-skill-group',open:selected});d.appendChild(el('summary',{},`<span><strong>${uxEsc(a.name)}</strong><small>Compétences liées à cet Attribut</small></span><span class="badge ${selected?'good':''}">${eligible.reduce((n,s)=>n+Number(state.skills[s.id].edge||0),0)} Edge</span>`));const grid=el('div',{class:'p32-five-grid'});eligible.forEach(sk=>{const o=state.skills[sk.id],total=typeof p25SkillTotal==='function'?p25SkillTotal(sk.id):skillRaw(sk.id),box=el('div',{class:'alloc p32-edge-skill'},`<div class="alloc-title"><strong>${uxEsc(sk.name)}</strong><span>${total}/5</span></div><div class="p26-card-help">${uxEsc(p25SkillHelp?.[sk.id]||'')}</div><div class="stepper"><button data-dir="-1">−</button><strong>+${o.edge}</strong><button data-dir="1">+</button></div>`);box.querySelectorAll('button').forEach(b=>b.onclick=()=>{const dir=Number(b.dataset.dir),nv=o.edge+dir;if(nv<0)return;if(dir>0&&(totalEdgeSkillPoints()>=budget||skillRaw(sk.id)>=5))return;o.edge=nv;render()});grid.appendChild(box)});d.appendChild(grid);c.appendChild(d)});
};

function p32EdgeTalentSelect(i){
  const groups=uxEdgeTalentGroups().map(([label,sub,items])=>({label:sub?`${label} — ${sub}`:label,items})),others=new Set(state.talents.edge.filter((_,j)=>j!==i).filter(Boolean));groups.forEach(g=>g.items=g.items.filter(t=>!others.has(t.id)));
  const field=el('section',{class:'p27-talent-section p32-edge-talent'}),value=state.talents.edge[i]||'',shell=el('div',{class:'field p27-select-field'});shell.appendChild(el('label',{},`Talent Edge ${i+1}`));const row=el('div',{class:'p27-select-shell'}),sel=el('select');sel.appendChild(el('option',{value:''},'— Choisir un Talent —'));
  groups.filter(g=>g.items.length).forEach(g=>{const og=el('optgroup',{label:g.label.toUpperCase()});g.items.forEach(t=>{const lore=p26TalentNarrative(t),tip=`Ambiance : ${lore}\nMécanique : ${t.effect||'—'}`,o=el('option',{value:t.id,title:tip},uxEsc(t.name));if(t.id===value)o.selected=true;og.appendChild(o)});sel.appendChild(og)});sel.onchange=e=>{state.talents.edge[i]=e.target.value;render()};row.appendChild(sel);const chosen=groups.flatMap(g=>g.items).find(t=>t.id===value)||talentById(value);if(chosen){const info=uxInfo(`Ambiance : ${p26TalentNarrative(chosen)}\nMécanique : ${chosen.effect||'—'}`);info.classList.add('p27-select-info');row.appendChild(info)}shell.appendChild(row);field.appendChild(shell);if(chosen)field.appendChild(p26TalentDetail(chosen));return field;
}
renderEdgeTalentChoices=function(c){trimEdgeTalentSlots();if(!state.edge.talentPacks)return;c.appendChild(el('div',{class:'section-title reality-subtitle'},'<div><h3>Choisir les Talents achetés avec Edge</h3><p>Le choix est effectué ici immédiatement. Le Talent d’Expertise acheté par Edge n’est pas limité aux deux familles du Style.</p></div>'));for(let i=0;i<state.edge.talentPacks;i++)c.appendChild(p32EdgeTalentSelect(i));};

stepEdge=function(){
  uxEnsureEdgeAttributes();if(hasDis('inconnu'))state.edge.renownPack=0;trimEdgeTalentSlots();const c=wrap('Edge','L’Edge représente la part de chance, de potentiel ou de ressources exceptionnelles qui distingue le personnage. Vous commencez avec 5 Edge, plus 1 par Désavantage ; ce qui n’est pas dépensé à la création reste disponible en jeu.',`<div class="kpi ${edgeRemaining()>=0?'good':'bad'}">${edgeRemaining()} restant / ${edgeTotal()}</div>`);
  if(!style())c.appendChild(el('div',{class:'rulebox warn'},'Choisissez un Style pour afficher correctement les valeurs économiques.'));
  const grid=el('div',{class:'p32-edge-spend-grid'});grid.append(
    p32EdgeSpendCard('+2 Attributs','attributePack',1,state.edge.attributePack?`${uxEdgeAttrTotal()}/2 points attribués`:'',()=>uxEdgeAttrTotal()===0),
    p32EdgeSpendCard('+4 Compétences','skillPacks',3,`${totalEdgeSkillPoints()} / ${edgeSkillBudget()} points attribués`,()=>totalEdgeSkillPoints()<=(state.edge.skillPacks-1)*4),
    p32EdgeSpendCard('+1 Talent de Réalité','talentPacks',2,`${state.talents.edge.filter(Boolean).length}/${state.edge.talentPacks} Talent(s) choisi(s)`,()=>state.talents.edge.slice(-1)[0]===''||state.talents.edge.length===0),
    p32EdgeSpendCard('+5 000 $ Compte','cashPacks',3,style()?`Compte de départ : ${fmt(style().account)} → ${fmt(startAccount())}`:''),
    p32EdgeSpendCard('+1 Train de vie','lifestylePack',1,style()?`Train de vie final actuel : ${lifestyleFinal()}`:'',()=>true,()=>lockLifestyleEdgeUseful()),
    p32EdgeSpendCard('+5 000 $ Aug. / Gen2','augmentationPacks',2,style()?`Enveloppe ${fmt(economic().envelope)} · ${gen2Windows()} fenêtre(s) Gen2`:''),
    p32EdgeSpendCard('+1 Renommée','renownPack',1,`Renommée actuelle : ${renown()}`,()=>true,()=>!hasDis('inconnu')&&!hasTalent('renomme'))
  );c.appendChild(grid);
  if(hasDis('inconnu'))c.appendChild(el('div',{class:'rulebox warn'},'<strong>Inconnu :</strong> Renommée initiale 0 et achat de Renommée interdit à la création.'));if(hasTalent('renomme'))c.appendChild(el('div',{class:'rulebox'},'<strong>Renommé :</strong> la Renommée initiale est déjà fixée à 2 ; dépenser un Edge ici serait sans effet.'));
  renderEdgeAttributeAllocator(c);renderEdgeSkillAllocator(c);renderEdgeTalentChoices(c);
  c.appendChild(el('div',{class:'rulebox'},`<strong>Edge conservé :</strong> ${edgeRemaining()} Edge restera disponible en jeu. Il peut notamment servir à Forcer le Destin ou Échapper au Destin. <strong>Edge ne peut jamais acheter de PTV.</strong><br><strong>Compte actuel après les achats déjà saisis :</strong> ${fmt(economic().account)}.`));
};
