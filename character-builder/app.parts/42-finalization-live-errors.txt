// Block 11 follow-up — keep visible error banners synchronized with live field validation.

// Harmonise the Corporatiste error wording with the free-text field actually shown in Finalisation.
const p42ErrorsBase=currentErrors;
currentErrors=function(){
  let errs=p42ErrorsBase().filter(x=>!x.startsWith('Appui Corporatiste :'));
  if(state.creation.sphere==='corporatiste'&&!String(state.reality?.sphereSupportDetail||'').trim()){
    errs.push('Appui Corporatiste : préciser concrètement l’Appui de Sphère (par exemple logement de fonction, couverture santé, transport ou autre avantage contractuel).');
  }
  return [...new Set(errs)];
};

function p42RefreshFinalErrors(){
  const c=$('#stepContent');if(!c||STEPS[stepIndex]?.[0]!=='finish')return;
  const errs=currentErrors(),blocking=errs.filter(x=>!x.startsWith('Attention :'));

  // Remove only the player-facing validation messages generated by Finalisation.
  [...c.querySelectorAll(':scope > .alert')].forEach(x=>x.remove());
  [...c.querySelectorAll(':scope > .rulebox')].forEach(x=>{
    if((x.textContent||'').trim().startsWith('Validation mécanique :'))x.remove();
  });

  const anchor=[...c.children].find(x=>x.classList?.contains('hr'))||c.querySelector('.p41-relations-grid')||c.querySelector('.p40-lore-box');
  let cursor=anchor;
  const insert=node=>{if(cursor?.after){cursor.after(node);cursor=node}else{c.appendChild(node);cursor=node}};

  errs.forEach(msg=>insert(el('div',{class:msg.startsWith('Attention :')?'alert warn':'alert'},uxEsc(msg))));
  if(!blocking.length)insert(el('div',{class:'rulebox p42-validation-ok'},'<strong>Validation mécanique :</strong> aucune erreur bloquante détectée par le builder.'));
}

const p42RefreshBase=p41RefreshValidationUI;
p41RefreshValidationUI=function(){
  p42RefreshBase();
  p42RefreshFinalErrors();
};
