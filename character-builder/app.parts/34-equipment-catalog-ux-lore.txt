function r34CategoryContext(item){

  if(item?.kind==='augmentation'){

    if(/optique|ocul|cyberoeil/.test(s+n))return 'CyberOptique : support ou module intégré à la vision. Selon le modèle, il remplace, complète ou instrumente directement ce que l’œil perçoit.';

    if(/interne|organe|cardio|respir|digest/.test(s+n))return 'Augmentation interne : l’appareil est intégré au fonctionnement organique du corps plutôt qu’ajouté comme simple équipement porté.';
    if(/derm|externe|peau|armure/.test(s+n))return 'Augmentation dermique ou externe : sa fonction passe par la peau, la protection corporelle ou une structure visible à la surface du corps.';
    if(/membre|bras|jambe|main|pied/.test(s+n))return 'Augmentation de membre : elle remplace ou transforme directement un bras, une jambe, une main ou un pied et dépend souvent d’un support adapté.';

  if(item?.vehicle)return 'Véhicule terrestre ou aérien : un bien durable dont le prix d’achat, la structure, le blindage, les places et l’entretien comptent autant que la simple vitesse.';
  if(/armure|protection|gallow|holor/.test(s+n))return 'Protection portée : elle est conçue pour encaisser un type de menace déterminé et doit être lue avec ses valeurs de protection plutôt qu’avec son seul prix.';

  if(/drogue|boisson|nourrit|repas|alcool/.test(s+n))return 'Produit de consommation : sa valeur est autant sociale et fictionnelle que technique, et les dépenses ordinaires peuvent être absorbées par le Train de vie.';

function r34ItemLore(item){
  if(!item)return '';
  const old=r34HistoricalWeapon(item);if(old)return old;
  const role=r34DeepText(item,['Rôle','Role']),type=r34DeepText(item,['Classe','Type']),appearance=r34DeepText(item,['Apparence','Aspect','Description physique','Présentation']),desc=r34DeepText(item,['Description','Principe']);
  if(appearance&&normText(appearance)!==normText(item.effect||''))return appearance;
  if(desc&&normText(desc)!==normText(item.effect||''))return desc;
  if(role)return `${type?`${type}. `:''}${role}. ${r34CategoryContext(item)}`;
  if(type&&!normText(item.category||'').includes(normText(type)))return `${type}. ${r34CategoryContext(item)}`;
  return r34CategoryContext(item);
}
