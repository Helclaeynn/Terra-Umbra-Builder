function t51FacetName(t){
  const parts=String(t?.group||'').split('›').map(x=>x.trim()).filter(Boolean),last=parts.at(-1)||'';
  return last.replace(/^Facette\s*:\s*/i,'').trim();
}

const t51LoreBase=t28TruthLore;
t28TruthLore=function(t){
  const key=normText(t?.name||'');
  if(t51SpecificLore[key])return t51SpecificLore[key];
  const old=t51LoreBase(t);
  return !old||t51MetaLorePattern.test(old)?t51InWorldFallback(t):old;
};

function t51LoreAudit(){
  const saved=state.truth.nature,problems=[];
  try{
    for(const [nature,rows] of Object.entries(truthCatalog||{})){
      if(!Array.isArray(rows))continue;state.truth.nature=nature;
      for(const t of rows){const lore=t28TruthLore(t)||'';if(t51MetaLorePattern.test(lore))problems.push({nature,name:t.name,group:t.group,lore})}
    }
  }finally{state.truth.nature=saved}
  return problems;
}
