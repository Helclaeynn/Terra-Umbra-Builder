function daSelectedFreeTraits(){
  const out=[],n=state.truth.nature,ch=state.truth.choices||{};
  if(n==='daemon'){
    const f=daFree.daemon.function[ch.function];if(f)out.push({...f,source:'Empreinte gratuite de Fonction'});
    const d=daFree.daemon.divinity[ch.divinity];if(d)out.push({...d,source:'Empreinte gratuite de Divinité'});
  }
  if(n==='angelus'){
    const a=daFree.angelus.nature[ch.angelNature];if(a)out.push({...a,source:'Pouvoir fondamental gratuit de Nature'});
    const s=daFree.angelus.sephirah[ch.sephirah];if(s)out.push({...s,source:'Don fondamental gratuit du Sephirah'});
  }
  return out;
}
