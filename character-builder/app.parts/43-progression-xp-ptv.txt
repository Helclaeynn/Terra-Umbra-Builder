function p43EnsureProgression(){
  const old=state.progression&&typeof state.progression==='object'?state.progression:{};
  state.progression={
    xpEarned:Math.max(0,Number(old.xpEarned)||0),
    ptvEarned:Math.max(0,Number(old.ptvEarned)||0),
    skillRanks:old.skillRanks&&typeof old.skillRanks==='object'?old.skillRanks:{},
    attributeRanks:old.attributeRanks&&typeof old.attributeRanks==='object'?old.attributeRanks:{},
    realityTalents:Array.isArray(old.realityTalents)?old.realityTalents:[],
    truthTalents:Array.isArray(old.truthTalents)?old.truthTalents:[],
    flashUses:Array.isArray(old.flashUses)?old.flashUses:[],
    flashReady:old.flashReady!==false,
    flashArmed:!!old.flashArmed
  };
  ruleset.skills.forEach(s=>state.progression.skillRanks[s.id]=Math.max(0,Math.floor(Number(state.progression.skillRanks[s.id])||0)));
  ruleset.attributes.forEach(a=>state.progression.attributeRanks[a.id]=Math.max(0,Math.floor(Number(state.progression.attributeRanks[a.id])||0)));
}

function p43SkillBase(id){return Number(skillRaw(id)||0)}
function p43SkillRanks(id){return Math.max(0,Number(state.progression.skillRanks[id])||0)}
function p43SkillRaw(id){return p43SkillBase(id)+p43SkillRanks(id)}
function p43AttributeBase(id){return Number(finalAttr(id)||0)}
function p43AttributeRanks(id){return Math.max(0,Number(state.progression.attributeRanks[id])||0)}
function p43AttributeValue(id){return p43AttributeBase(id)+p43AttributeRanks(id)}
function p43FlashKey(id,step){return `${id}:${step}`}
function p43OwnRealityTalent(id){return hasTalent(id)||(state.progression.realityTalents||[]).includes(id)}
function p43SkillStepBaseCost(id,step){const from=p43SkillBase(id)+step-1;return from===0?3:from+1}
function p43SkillStepCost(id,step){const base=p43SkillStepBaseCost(id,step);return state.progression.flashUses.includes(p43FlashKey(id,step))?Math.max(1,base-2):base}
function p43SkillSpent(id){let n=0;for(let i=1;i<=p43SkillRanks(id);i++)n+=p43SkillStepCost(id,i);return n}
function p43AttributeStepCost(id,step){return 3*(p43AttributeBase(id)+step)}
function p43AttributeSpent(id){let n=0;for(let i=1;i<=p43AttributeRanks(id);i++)n+=p43AttributeStepCost(id,i);return n}
function p43XpSpent(){return ruleset.skills.reduce((n,s)=>n+p43SkillSpent(s.id),0)+ruleset.attributes.reduce((n,a)=>n+p43AttributeSpent(a.id),0)+(state.progression.realityTalents||[]).length*10}
function p43XpRemaining(){return Math.max(0,Number(state.progression.xpEarned)||0)-p43XpSpent()}
function p43PtvSpent(){return (state.progression.truthTalents||[]).reduce((n,id)=>n+(truthTalentById(id)?.cost||0),0)}
function p43CreationPtvReserve(){return Math.max(0,truthPtvRemaining())}
function p43PtvRemaining(){return p43CreationPtvReserve()+Math.max(0,Number(state.progression.ptvEarned)||0)-p43PtvSpent()}
function p43CanSpendXp(cost){return p43XpRemaining()>=cost}
function p43CanSpendPtv(cost){return p43PtvRemaining()>=cost}
function p43ProgressTalentBonus(skillId){return (state.progression.realityTalents||[]).reduce((n,id)=>n+(p20SkillTalentMap?.[id]===skillId?1:0),0)}
function p43SkillFinal(id){return p43SkillRaw(id)+(typeof p20SkillTalentBonus==='function'?p20SkillTalentBonus(id):0)+p43ProgressTalentBonus(id)}

function p43SpendSkill(id){
  const ranks=p43SkillRanks(id);if(p43SkillBase(id)+ranks>=15)return;
  const step=ranks+1,key=p43FlashKey(id,step),baseCost=p43SkillStepBaseCost(id,step),useFlash=p43OwnRealityTalent('apprentissage_fulgurant')&&state.progression.flashReady&&state.progression.flashArmed;
  const cost=useFlash?Math.max(1,baseCost-2):baseCost;if(!p43CanSpendXp(cost))return;
  state.progression.skillRanks[id]=step;
  if(useFlash){state.progression.flashUses.push(key);state.progression.flashReady=false;state.progression.flashArmed=false}
  render();
}
function p43RefundSkill(id){
  const ranks=p43SkillRanks(id);if(!ranks)return;const key=p43FlashKey(id,ranks);
  state.progression.skillRanks[id]=ranks-1;
  if(state.progression.flashUses.includes(key)){state.progression.flashUses=state.progression.flashUses.filter(x=>x!==key);state.progression.flashReady=true}
  render();
}
function p43SpendAttribute(id){const ranks=p43AttributeRanks(id);if(p43AttributeBase(id)+ranks>=10)return;const cost=p43AttributeStepCost(id,ranks+1);if(!p43CanSpendXp(cost))return;state.progression.attributeRanks[id]=ranks+1;render()}
function p43RefundAttribute(id){const ranks=p43AttributeRanks(id);if(!ranks)return;state.progression.attributeRanks[id]=ranks-1;render()}

function p43CreationRealityIds(){return new Set([state.talents.origin,state.talents.sphere,state.talents.expertise,state.talents.common,...(state.talents.edge||[])].filter(Boolean))}
function p43RealityTalentAllowed(t){
  if(!t)return {ok:false,reason:'Talent introuvable'};
  if(p43CreationRealityIds().has(t.id)||(state.progression.realityTalents||[]).includes(t.id))return {ok:false,reason:'Déjà acquis'};
  if(t.id==='neurodriver'&&(p43SkillRaw('neurodive')<1||hasDis('unsinkable')))return {ok:false,reason:'Neurodive 1+ requis et incompatible avec Unsinkable'};
  if(t.id==='brave'&&hasDis('lache'))return {ok:false,reason:'Incompatible avec Lâche'};
  if(t.id==='resistance_a_la_chaleur'&&hasDis('sensible_a_la_chaleur'))return {ok:false,reason:'Incompatible avec Sensible à la chaleur'};
  if(t.id==='resistance_au_froid'&&hasDis('sensible_au_froid'))return {ok:false,reason:'Incompatible avec Sensible au froid'};
  return {ok:true};
}
