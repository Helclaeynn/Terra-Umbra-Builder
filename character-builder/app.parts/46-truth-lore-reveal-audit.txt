function t46TruthActionLore(t){
  const e=normText(t.effect||''),g=String(t.group||''),tail=(g.split('›').map(x=>x.trim()).filter(Boolean).slice(-1)[0]||g||'cette tradition');
  if(/soin|pv|regener|restaur|gueri/.test(e))return `L’apprentissage donne à ${tail} une manière concrète de préserver ce qui devrait céder : la récupération décrite par la règle devient un geste, un réflexe ou une discipline reconnaissable dans sa tradition.`;
  if(/furtiv|invisib|dissimul|ombre/.test(e))return `La technique enseigne à disparaître de l’attention avant de disparaître du regard. Dans ${tail}, posture, rythme et pouvoir se combinent jusqu’à rendre l’approche surnaturellement difficile à suivre.`;
  if(/perception|detect|voir|vision|sentir|lecture/.test(e))return `Ce Talent change ce que le personnage considère comme une information évidente. ${tail} lui apprend à remarquer un signal que d’autres ne possèdent simplement pas les sens, l’instinct ou la formation nécessaires pour lire.`;
  if(/dgt|attaque|frapp|degat|arme|pugilat/.test(e))return `Ici, ${tail} devient une doctrine de violence précise : le personnage n’est pas seulement plus dangereux, il sait exactement dans quelles conditions sa Nature peut transformer une attaque ordinaire en manifestation de Vérité.`;
  if(/defense|armure|protection|resist/.test(e))return `Ce savoir est né de situations où survivre exige davantage qu’encaisser. ${tail} fournit une réponse surnaturelle ou doctrinale à l’agression, avec les limites précises décrites par l’effet mécanique.`;
  if(/pa|initiative|deplacement|bond|vitesse|reaction/.test(e))return `Le Talent modifie le tempo plutôt que la puissance brute. Dans ${tail}, agir au bon instant, franchir l’espace ou conserver l’initiative fait partie de la maîtrise elle-même.`;
  if(/social|charisme|diplom|influence|autorite|convain/.test(e))return `La tradition ne remplace pas la personnalité du personnage : elle lui apprend quel poids sa Nature peut donner à une présence, une parole ou un lien lorsque la scène sociale correspond réellement à son enseignement.`;
  if(/savoir|memoire|connaissance|information|question|mj/.test(e))return `Ce Talent transforme l’expérience accumulée par ${tail} en méthode fiable. Il ne donne pas l’omniscience : il rend accessible un type précis de connaissance que le personnage a appris à chercher, conserver ou interpréter.`;
  if(/peur|terreur|panique|frenesie|rage/.test(e))return `L’enseignement travaille directement avec les réflexes les plus anciens : peur, rage ou instinct. ${tail} apprend au personnage à provoquer, contenir ou chevaucher cette impulsion sans la confondre avec une simple émotion ordinaire.`;
  if(/rituel|magie|aura|spectr|occulte|voile|hologram/.test(e))return `Le personnage apprend où ${tail} touche réellement aux lois cachées du monde. Le Talent matérialise cette compréhension par une permission occulte précise plutôt que par une magie vague et universelle.`;
  return `Dans ${tail}, ce Talent correspond à une pratique identifiable, transmise et exercée. Son effet mécanique décrit le résultat ; en fiction, l’acquérir signifie que le personnage a réellement intégré cette manière particulière d’exprimer sa Vérité.`;
}
