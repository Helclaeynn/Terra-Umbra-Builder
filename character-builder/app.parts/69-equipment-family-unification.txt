function r69WeaponFamily(item){
  if(!item||item.kind==='augmentation'||item.neuro)return '';
  const name=r34Loose(item.name||''),legacy=r69LegacyWeaponFamilies.get(name);if(legacy)return legacy;
  const category=r34Loose(item.category||''),weaponClass=r34Loose(r15DeepField(item,['Classe','Type'])||''),role=r34Loose(r15DeepField(item,['Rôle','Role'])||'');
  if(category==='armes melee'){
    if(/trait|jet/.test(weaponClass)||/\b(?:lc|hl|jl|th|mb) \d/.test(name))return 'Armement — Armes de jet & trait';
    return 'Armement — Armes de mêlée';
  }
  if(category==='armes poing tasers'){
    if(/taser/.test(weaponClass)||/\b(?:lt|ht) \d/.test(name))return 'Armement — Tasers';
    if(weaponClass==='poche'||/\bpp \d/.test(name))return 'Armement — Pistolets de poche';
    if(/pistolet leger/.test(weaponClass)||/\blp \d/.test(name))return 'Armement — Pistolets légers';
    if(/pistolet lourd/.test(weaponClass)||/\bhp \d/.test(name))return 'Armement — Pistolets lourds';
  }
  if(category==='armes automatiques'){
    if(weaponClass==='pm'||/\bmgp \d/.test(name))return 'Armement — Pistolets-mitrailleurs (PM)';
    if(weaponClass==='smg'||/\blmg \d/.test(name))return 'Armement — SMG';
    if(/assaut/.test(weaponClass)||/\bar \d/.test(name))return 'Armement — Fusils d’assaut';
  }
  if(category==='armes precision'){
    if(/shotgun/.test(weaponClass)||/\bsg \d/.test(name))return 'Armement — Shotguns';
    return 'Armement — Fusils de précision';
  }
  if(category==='armes lourdes'){
    if(/mitrailleuse|gatling/.test(role)||/bastion|hellstorm/.test(name))return 'Armement — Mitrailleuses lourdes & Gatlings';
    if(/neutralisation/.test(role)||/manticore/.test(name))return 'Armement — Neutralisation lourde';
    if(/lance grenades|lance roquette|missile/.test(role)||/doorbell|breacher|wasp/.test(name))return 'Armement — Lanceurs & missiles';
    if(/anti materiel|fortification/.test(role)||/wallbreaker/.test(name))return 'Armement — Anti-matériel & fortifications';
    if(/projecteur/.test(role)||/salamander|purifier/.test(name))return 'Armement — Projecteurs lourds';
    return 'Armement — Armes lourdes';
  }
  return '';
}

const r69DisplayCategoryBase=eqxDisplayCategory;
eqxDisplayCategory=function(item){
  if(item?.neuro)return 'Neuroprogrammes';
  const weapon=r69WeaponFamily(item);if(weapon)return weapon;
  return r69DisplayCategoryBase(item);
};

  const item=(realityData.equipment||[]).find(x=>r34Loose(x.name||'')===r34Loose(name||''));
