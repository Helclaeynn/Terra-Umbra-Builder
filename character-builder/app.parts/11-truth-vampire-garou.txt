function vgSelectedFreeTraits(){
  const nature=state.truth.nature,traits=[];
  if(nature==='vampire'){
    const t=vgFreeChoiceTraits.vampire.court[state.truth.choices?.court];if(t)traits.push({...t,source:'Empreinte gratuite de Cour'});
  }
  if(nature==='garou'){
    const p=vgFreeChoiceTraits.garou.pelage[state.truth.choices?.pelage];if(p)traits.push({...p,source:'Tradition gratuite de Pelage'});
    const b=vgFreeChoiceTraits.garou.blood[state.truth.choices?.blood];if(b)traits.push({...b,source:'Règle native du Sang vif'});
  }
  return traits;
}
const vgBaseStepTruth=stepTruth;
stepTruth=function(){
  vgBaseStepTruth();
  const traits=vgSelectedFreeTraits();if(!traits.length)return;
  const c=$('#stepContent');if(!c)return;
  const section=el('div',{class:'vg-free-traits'});section.appendChild(el('h3',{class:'subhead'},'Capacités gratuites liées aux choix'));
  const grid=el('div',{class:'trait-grid'});
  traits.forEach(t=>grid.appendChild(el('div',{class:'trait'},`<div class="trait-head"><strong>${t.name}</strong><span class="chip">${t.access}</span></div><div class="skill-meta">${t.source} · 0 PTV</div><p>${t.effect}</p>`)));
  section.appendChild(grid);c.insertBefore(section,c.querySelector('.truth-list')||null);
};
