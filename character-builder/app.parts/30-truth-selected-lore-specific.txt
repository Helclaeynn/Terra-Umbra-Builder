// Structural Truth choices: hover helps BEFORE selection; once selected, show the lore directly below.
function t30TruthChoiceSelect(ch){
  const value=state.truth.choices?.[ch.key]||'';
  const field=el('div',{class:'field t30-truth-choice'}),lab=el('label',{},`${ch.label}${ch.optional?' · optionnel':''}`),select=el('select');
  select.appendChild(el('option',{value:''},ch.optional?'— Aucun —':'— Choisir —'));
  (ch.options||[]).forEach(o=>{
    const attrs={value:o.id};
    // The selected value is explained below instead of relying on a tooltip.
    if(o.id!==value && o.description)attrs.title=o.description;
    const opt=el('option',attrs,uxEsc(o.name));if(o.id===value)opt.selected=true;select.appendChild(opt);
  });
  select.onchange=e=>t16ChoiceChanged(ch.key,e.target.value);field.append(lab,select);
  const picked=(ch.options||[]).find(o=>o.id===value);
  if(picked?.description){
    const detail=el('div',{class:'t30-choice-detail'},`<strong>${uxEsc(picked.name)}</strong><em>${uxEsc(picked.description)}</em>`);
    field.appendChild(detail);
  }
  return field;
}
t16RenderChoicePanel=function(){
  const n=truthNature(),old=$('#stepContent .truth-nature-card')?.nextElementSibling;if(!n?.choices?.length)return;
  const panel=el('div',{class:'t16-choice-stack t30-choice-stack'});n.choices.forEach(ch=>panel.appendChild(t30TruthChoiceSelect(ch)));
  if(old?.classList?.contains('grid2')||old?.classList?.contains('t16-choice-stack'))old.replaceWith(panel);else $('#stepContent .truth-nature-card')?.after(panel);
};

function t30DaemonDivinity(t){const g=normText(t.group||'');return Object.keys(t30DivinityDomains).find(k=>g.includes(normText(k)))||''}
function t30EffectLore(t){
  const name=t.name||'Ce Talent',effect=normText(t.effect||''),group=t.group||'';
  if(effect.includes('souvenir')||effect.includes('savoir')||effect.includes('information'))return `« ${name} » transforme une connaissance en instinct de Vérité : ce que le personnage a appris cesse d’être seulement théorique et devient immédiatement exploitable lorsque la situation l’exige.`;
  if(effect.includes('soin')||effect.includes('recuper')||effect.includes('pv'))return `« ${name} » donne à la puissance surnaturelle une fonction de survie : chair, âme ou énergie sont forcées de tenir là où un corps ordinaire commencerait à céder.`;
  if(effect.includes('defense')||effect.includes('armure')||effect.includes('protection'))return `« ${name} » matérialise une volonté de ne pas céder : la Vérité du personnage devient rempart, détourne l’impact ou impose à l’agression une résistance qu’elle n’attendait pas.`;
  if(effect.includes('peur')||effect.includes('tendu')||effect.includes('panique'))return `« ${name} » agit sur l’instant où l’instinct comprend le danger avant la raison : la présence surnaturelle devient une menace que le corps de la cible croit avant même de l’expliquer.`;
  if(effect.includes('ordre')||effect.includes('autorite')||effect.includes('social'))return `« ${name} » donne du poids surnaturel à une parole, un statut ou une intention ; l’influence ne vient plus seulement du charisme, mais de ce que la Vérité fait résonner derrière les mots.`;
  if(effect.includes('illusion')||effect.includes('perception')||effect.includes('vision'))return `« ${name} » travaille la frontière entre ce qui existe et ce qui est perçu : les sens reçoivent une information que la Réalité seule n’aurait jamais pu fournir.`;
  if(effect.includes('deplacement')||effect.includes('mouvement')||effect.includes('teleport')||effect.includes('poursu'))return `« ${name} » transforme le déplacement en expression de Nature : distance, trajectoire ou poursuite cessent d’obéir tout à fait aux limitations ordinaires du corps.`;
  if(effect.includes('dgt')||effect.includes('attaque')||effect.includes('degat'))return `« ${name} » donne une forme offensive directe à la Vérité du personnage : ce qui n’était jusque-là qu’un héritage ou une affinité devient une menace tangible sur le champ de bataille.`;
  if(effect.includes('zone')||effect.includes('rayon'))return `« ${name} » cesse d’être une manifestation intime : le pouvoir déborde du porteur et impose temporairement sa logique à tout un espace autour de lui.`;
  if(effect.includes('prerequis'))return `« ${name} » représente un degré de maîtrise qui ne peut apparaître qu’après avoir appris à contrôler la forme plus simple dont il procède.`;
  return `« ${name} » est une pratique distincte de ${group||'Vérité'} : son apprentissage marque une manière précise dont le personnage a appris à faire passer sa Nature dans le monde.`;
}

// Replace only the repetitive generic lore; keep all hand-authored Hunter lore from V7.
const t30LoreBase=t28TruthLore;
t28TruthLore=function(t){
  const n=state.truth.nature,key=normText(t.name||'');
  if(n==='daemon'){
    if(t30DaemonLore[key])return t30DaemonLore[key];
    const god=t30DaemonDivinity(t);if(god){
      const domain=t30DivinityDomains[god];
      const effectLore=t30EffectLore(t);
      return `${effectLore} Chez un Élu de ${god.charAt(0).toUpperCase()+god.slice(1)}, cette expression reste enracinée dans ${domain}.`;
    }
    return t30EffectLore(t);
  }
  const old=t30LoreBase(t),generic=old?.startsWith(`« ${t.name||''} »`);
  return generic?t30EffectLore(t):old;
};
